function tIcon(n){ return (window.ICONS && ICONS[n]) || iconFor(n) || ""; }
function tStatus(n){ return isV(n) ? "vis" : isU(n) ? "unl" : available(n) ? "reach" : "locked"; }
function tPath(a, b){
  const [x1,y1] = TPOS[a], [x2,y2] = TPOS[b]; const r = 30;
  const dx = x2 - x1, dy = y2 - y1, d = Math.hypot(dx, dy) || 1;
  const sx = x1 + dx / d * r, sy = y1 + dy / d * r, ex = x2 - dx / d * (r + 6), ey = y2 - dy / d * (r + 6);
  const bend = Math.min(60, d * .18), nx = -dy / d * bend, ny = dx / d * bend;
  return `M${sx.toFixed(1)},${sy.toFixed(1)} Q${((sx + ex) / 2 + nx).toFixed(1)},${((sy + ey) / 2 + ny).toFixed(1)} ${ex.toFixed(1)},${ey.toFixed(1)}`;
}

function renderTree(){
  const host = $("#tree"); if (!host) return;
  if (!tv) tv = [0, 0, 2000, 1150];
  const stage = $(".tree-stage", host);
  stage.querySelector("svg")?.remove();
  stage.insertAdjacentHTML("afterbegin", treeSvg());
  $(".tree-side", host).innerHTML = tsel ? treeSide(tsel) : `<p>Click a moon.</p>`;
  const svg = $("#treesvg");
  $$(".node", svg).forEach(g => {
    const pick = () => { tsel = g.dataset.n; renderTree(); };
    g.addEventListener("click", () => { if (!svg.dataset.moved) pick(); });
    g.addEventListener("dblclick", () => { const m = M[g.dataset.n]; if (m?.slug) location.hash = m.slug; });
    g.addEventListener("keydown", e => { if (e.key === "Enter" || e.key === " ") { e.preventDefault(); pick(); } });
  });
  const su = $("#tsu"), sv = $("#tsv");
  if (su) su.onclick = () => setU(tsel, !isU(tsel));
  if (sv) sv.onclick = () => setV(tsel, !isV(tsel));
  wirePanZoom(svg);
}
function wirePanZoom(svg){
  const stage = svg.parentElement;
  const apply = () => svg.setAttribute("viewBox", tv.join(" "));
  const scaleOf = () => { const r = svg.getBoundingClientRect(); return Math.max(tv[2] / r.width, tv[3] / r.height); };
  const toSvg = (cx, cy) => { const r = svg.getBoundingClientRect(); const s = scaleOf(); const ox = (r.width * s - tv[2]) / 2, oy = (r.height * s - tv[3]) / 2; return [tv[0] - ox + (cx - r.left) * s, tv[1] - oy + (cy - r.top) * s]; };
  const zoom = (f, cx, cy) => { const [px, py] = cx == null ? [tv[0] + tv[2] / 2, tv[1] + tv[3] / 2] : toSvg(cx, cy); const w = Math.min(2600, Math.max(300, tv[2] * f)); const k = w / tv[2]; tv = [px - (px - tv[0]) * k, py - (py - tv[1]) * k, w, tv[3] * k]; apply(); };
  svg.addEventListener("wheel", e => { e.preventDefault(); zoom(e.deltaY > 0 ? 1.15 : 1 / 1.15, e.clientX, e.clientY); }, {passive: false});
  let drag = null;
  svg.addEventListener("pointerdown", e => { drag = {x: e.clientX, y: e.clientY, v: tv.slice(), s: scaleOf()}; delete svg.dataset.moved; });
  svg.addEventListener("pointermove", e => { if (!drag) return; const dx = e.clientX - drag.x, dy = e.clientY - drag.y; if (Math.abs(dx) + Math.abs(dy) > 4){ svg.dataset.moved = 1; stage.classList.add("drag"); } tv = [drag.v[0] - dx * drag.s, drag.v[1] - dy * drag.s, drag.v[2], drag.v[3]]; apply(); });
  const end = () => { drag = null; stage.classList.remove("drag"); setTimeout(() => delete svg.dataset.moved, 0); };
  svg.addEventListener("pointerup", end); svg.addEventListener("pointerleave", end);
  $("#tzin").onclick = () => zoom(1 / 1.3); $("#tzout").onclick = () => zoom(1.3);
  $("#tzreset").onclick = () => { tv = [0, 0, 2000, 1150]; apply(); };
  $("#tall").onchange = e => { tAll = e.target.checked; renderTree(); };
}
